Copy these values into the Hostinger Node.js environment variables or into the root .env file:

NODE_ENV=production
PORT=3000
APP_URL=https://YOUR-DOMAIN.com
CLIENT_URL=https://YOUR-DOMAIN.com
DB_HOST=localhost
DB_PORT=3306
DB_NAME=YOUR_HOSTINGER_DATABASE_NAME
DB_USER=YOUR_HOSTINGER_DATABASE_USER
DB_PASSWORD=YOUR_HOSTINGER_DATABASE_PASSWORD
DB_SSL=false
JWT_SECRET=GENERATE_A_LONG_RANDOM_SECRET
JWT_EXPIRES_IN=12h
ADMIN_EMAIL=admin@YOUR-DOMAIN.com
ADMIN_PASSWORD=SET_A_STRONG_INITIAL_PASSWORD
UPLOAD_DIR=uploads
MAX_UPLOAD_MB=20
